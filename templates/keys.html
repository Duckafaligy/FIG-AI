The calls a partner makes

# provision a site when their customer turns the feature on
curl -X POST {{ public_url }}/v1/sites \
  -H "X-API-Key: $FIG_KEY" -H "Content-Type: application/json" \
  -d '{"hostname":"clientdomain.com","client_name":"Their Client Ltd"}'

# read it
curl -X POST {{ public_url }}/v1/sites/$SITE_ID/scans -H "X-API-Key: $FIG_KEY"

# pull the result into their own UI
curl {{ public_url }}/v1/scans/$SCAN_ID -H "X-API-Key: $FIG_KEY"

# or the whole estate as one report, worst first
curl {{ public_url }}/v1/report -H "X-API-Key: $FIG_KEY"
